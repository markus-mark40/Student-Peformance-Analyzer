// Initialize START
SET studentNumbers to 0
SET currentStudentNumber to 1
SET studentScore to -1

SET totalScore to 0
SET averageScore to 0

SET highestScore to 0
SET lowestScore to 0

SET passingCount to 0
SET failingCount to 0

SET scoreCategory to none
// Initialize END

// Ask for studentNumbers
WHILE studentNumbers is not greater than 0 THEN
    ask for studentNumbers

    IF studentNumbers is not greater than 0 THEN
        say "Number is invalid. Number must be greater than 0"

// After the studentNumber is now greater than 0

// Main comparison block START
WHILE currentStudentNumber is less than or equal to studentNumbers THEN
    ask for studentScore

    // Repeats on asking for a valid studentScore
    while studentScore is less than 0 OR studentScore is greater than 100 THEN
        if studentScore is less than 0 OR studentScore is greater than 100 THEN
            DISPLAY "Number is invalid. Number must be  greater than or equal to 0 and less than or equal to 100"
        ask for studentScore

    // Categorize the studentScores
    IF studentScore is 90-100 THEN
        SET scoreCategory as "Excellent"
    ELSE IF studentScore is 80-89 THEN
        SET scoreCategory as "Very Good"
    ELSE IF studentScore is 75-79 THEN
        SET scoreCategory as "Good"
    ELSE IF studentScore is 60-74 THEN
        SET scoreCategory as "Needs Improvement"
    ELSE IF studentScore is 59 below THEN
        SET scoreCategory as "Failing"

    // Initialize currentStudentNumber to thefirst studentScore
    IF currentStudentNumber is equal to 1 THEN
        SET highestScore to studentScore
        SET lowestScore to studentScore
    ELSE // Sets the highestScore and lowestScore
        IF studentScore is greater than highestScore THEN
            SET highestScore to studentScore
        IF studentScore is less than lowestScore THEN
            SET lowestScore to studentScore

    // Calculates and sets the totalScore
    SET totalScore to totalScore + studentScore
    
    // Counts the passingCount and failignCount
    IF studentScore is greater than or equal to 60 THEN
        Add 1 to passingCount
    ELSE
        Add 1 to failingCount

    // Displays the iteration's results
    DISPLAY currentStudentNumber
    DISPLAY studentScore
    DISPLAY "Valid"
    display scoreCategory
    DISPLAY passingCount
    DISPLAY failingCount
    DISPLAY totalScore

    add 1 to the currentStudentNumber after comparing
// Main comparison block END

// calculate averageScore
SET averageScore to totalScore / studentNumbers

//Display the summary
DISPLAY averageScore
DISPLAY passingCount
DISPLAY failingCount
DISPLAY highestScore
DISPLAY lowestScore
DISPLAY totalScore
