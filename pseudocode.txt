// Initialize variables
studentNumbers = 0
currentStudentNumber = 1
studentScore = -1

totalScore = 0
averageScore = 0

highestScore = 0
lowestScore = 0

passingCount = 0
failingCount = 0

// Ask for studentNumbers and repeats asking until a valid number is given.
while (studentNumbers <= 0)
{
    input studentNumbers

    if (studentNumbers <= 0)
    then {
        print "Number is invalid. Number must be greater than 0"
    }
}

// Main comparison block START
while (currentStudentNumber <= studentNumbers)
{
    input studentScore

    // Repeats on asking for a valid studentScore
    while (studentScore < 0 or studentScore > 100)
    {
        if (studentScore < 0 or studentScore > 100)
        then {
            print "Number is invalid. Number must be greater than or equal to 0 and less than or equal to 100"
        }
        
        input studentScore
    }

    // Categorize studentScores
    if (studentScore >= 90)
    then {
        scoreCategory = "Excellent"
    } else {
        if (studentScore >= 80 and studentScore <= 89)
        then {
            scoreCategory = "Very Good"
        } else {
            if (studentScore >= 75 and studentScore <= 79)
            then {
                scoreCategory = "Good"
            } else {
                if (studentScore >= 60 and studentScore <= 74)
                then {
                    scoreCategory = "Needs Improvement"
                } else {
                    scoreCategory = "Failing"
                }
            }
        }
    }

    // Initialize highestScore and lowestScore to the first studentScore
    // Compares and sets the highestScore and lowestScore
    if (currentStudentNumber = 1)
    then{
        highestScore = studentScore
        lowestScore = studentScore
    } else {
        if (studentScore > highestScore)
        then {
            highestScore = studentScore
        }
        if (studentScore < lowestScore){
        lowestScore = studentScore
        }
    }

    // Calculates and sets the totalScore
    totalScore = totalScore + studentScore

    // Compares and counts the passingCount and failingCount
    if (studentScore >= 60)
    then {
        passingCount = passingCount + 1
    } else {
        failingCount = failingCount + 1
    }

    // Displays the current iteration's results
    Output currentStudentNumber
    Output studentScore
    Output "Valid"
    Output scoreCategory
    Output passingCount
    Output failingCount
    Output totalScore

    // Adds 1 to the currentStudentNumber
    currentStudentNumber = currentStudentNumber + 1
}

// Calculates and sets averageScore
averageScore = totalScore / studentNumbers

// Outputs the analysis summary
Output averageScore
Output passingCount
Output failingCount
Output highestScore
Output lowestScore
Output totalScore
